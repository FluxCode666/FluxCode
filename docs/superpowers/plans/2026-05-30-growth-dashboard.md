```go
func ParseGrowthQueryRange(startDate, endDate, granularity string, now time.Time) (GrowthQueryRange, error) {
	loc := growthLocation()
	if now.IsZero() {
		now = time.Now()
	}
	localNow := now.In(loc)
	if startDate == "" {
		startDate = time.Date(localNow.Year(), localNow.Month(), localNow.Day(), 0, 0, 0, 0, loc).AddDate(0, 0, -29).Format("2006-01-02")
	}
	if endDate == "" {
		endDate = localNow.Format("2006-01-02")
	}
	g := GrowthGranularity(granularity)
	if g == "" {
		g = GrowthGranularityDay
	}
	if g != GrowthGranularityDay && g != GrowthGranularityWeek && g != GrowthGranularityMonth {
		return GrowthQueryRange{}, fmt.Errorf("%w: invalid granularity", ErrGrowthInvalidQueryRange)
	}
	start, err := time.ParseInLocation("2006-01-02", startDate, loc)
	if err != nil {
		return GrowthQueryRange{}, fmt.Errorf("%w: invalid start_date", ErrGrowthInvalidQueryRange)
	}
	endDay, err := time.ParseInLocation("2006-01-02", endDate, loc)
	if err != nil {
		return GrowthQueryRange{}, fmt.Errorf("%w: invalid end_date", ErrGrowthInvalidQueryRange)
	}
	end := endDay.AddDate(0, 0, 1)
	if !end.After(start) {
		return GrowthQueryRange{}, fmt.Errorf("%w: start_date must be before or equal to end_date", ErrGrowthInvalidQueryRange)
	}
	if int(end.Sub(start).Hours()/24) > GrowthMaxRangeDays {
		return GrowthQueryRange{}, fmt.Errorf("%w: range exceeds %d days", ErrGrowthInvalidQueryRange, GrowthMaxRangeDays)
	}
	return GrowthQueryRange{
		Start:       start,
		End:         end,
		StartDate:   start.Format("2006-01-02"),
		EndDate:     end.Add(-time.Nanosecond).In(loc).Format("2006-01-02"),
		Granularity: g,
	}, nil
}

func growthLocation() *time.Location {
	loc, err := time.LoadLocation(GrowthLocationName)
	if err != nil {
		return time.FixedZone(GrowthLocationName, 8*60*60)
	}
	return loc
}

func growthSafeRate(numerator, denominator int64) float64 {
	if denominator <= 0 {
		return 0
	}
	return growthRound4(float64(numerator) / float64(denominator))
}

func growthRound2(v float64) float64 {
	if math.IsNaN(v) || math.IsInf(v, 0) {
		return 0
	}
	return math.Round(v*100) / 100
}

func growthRound4(v float64) float64 {
	if math.IsNaN(v) || math.IsInf(v, 0) {
		return 0
	}
	return math.Round(v*10000) / 10000
}
```

```go
	if s == nil || s.repo == nil {

	loc := growthLocation()

	todayStart := time.Date(localNow.Year(), localNow.Month(), localNow.Day(), 0, 0, 0, 0, loc)
	todayEnd := todayStart.AddDate(0, 0, 1)
	monthStart := time.Date(localNow.Year(), localNow.Month(), 1, 0, 0, 0, 0, loc)
	monthEnd := monthStart.AddDate(0, 1, 0)

	out, err := s.repo.GetOverview(ctx, r, todayStart, todayEnd, monthStart, monthEnd)

		return nil, err
	}
	if out == nil {
		return &GrowthOverview{}, nil
	}
	out.MonthRevenue = growthRound2(out.MonthRevenue)
	out.ARPU = growthRound2(out.ARPU)
	out.PaymentConversionRate = growthRound4(out.PaymentConversionRate)
	out.RepurchaseRate = growthRound4(out.RepurchaseRate)
	return out, nil
}
```

```go
package repository

import (
	"context"
	"database/sql"
	"fmt"
	"time"

	"github.com/Wei-Shaw/sub2api/internal/service"
)

type growthRepository struct {
	db *sql.DB
}

func NewGrowthRepository(db *sql.DB) service.GrowthRepository {
	return &growthRepository{db: db}
}

func (r *growthRepository) GetOverview(ctx context.Context, qr service.GrowthQueryRange, todayStart, todayEnd, monthStart, monthEnd time.Time) (*service.GrowthOverview, error) {
	if r == nil || r.db == nil {
		return &service.GrowthOverview{}, nil
	}

	var out service.GrowthOverview
	err := r.db.QueryRowContext(ctx, `
WITH active_range AS (
  SELECT COUNT(DISTINCT user_id) AS active_users
  FROM usage_logs
  WHERE created_at >= $1 AND created_at < $2
),
paid_range AS (
  SELECT COUNT(DISTINCT user_id) AS paid_users
  FROM payment_orders
  WHERE status IN ('PAID','RECHARGING','COMPLETED') AND paid_at >= $1 AND paid_at < $2
),
repurchase_range AS (
  SELECT COUNT(*) AS repurchase_users
  FROM (
    SELECT user_id
    FROM payment_orders
    WHERE status IN ('PAID','RECHARGING','COMPLETED') AND paid_at >= $1 AND paid_at < $2
    GROUP BY user_id
    HAVING COUNT(*) >= 2
  ) t
)
SELECT
  (SELECT COUNT(*) FROM users WHERE deleted_at IS NULL) AS total_users,
  (SELECT COUNT(DISTINCT user_id) FROM usage_logs WHERE created_at >= $3 AND created_at < $4) AS dau,
  (SELECT COUNT(DISTINCT user_id) FROM usage_logs WHERE created_at >= $5 AND created_at < $6) AS mau,
  (SELECT COUNT(*) FROM users WHERE deleted_at IS NULL AND created_at >= $3 AND created_at < $4) AS today_new_users,
  (SELECT COUNT(DISTINCT user_id) FROM payment_orders WHERE status IN ('PAID','RECHARGING','COMPLETED') AND paid_at >= $3 AND paid_at < $4) AS today_paid_users,
  COALESCE((SELECT SUM(pay_amount) FROM payment_orders WHERE status IN ('PAID','RECHARGING','COMPLETED') AND paid_at >= $5 AND paid_at < $6), 0) AS month_revenue,
  CASE WHEN (SELECT active_users FROM active_range) > 0
    THEN COALESCE((SELECT SUM(pay_amount) FROM payment_orders WHERE status IN ('PAID','RECHARGING','COMPLETED') AND paid_at >= $1 AND paid_at < $2), 0) / (SELECT active_users FROM active_range)
    ELSE 0 END AS arpu,
  CASE WHEN (SELECT active_users FROM active_range) > 0
    THEN (SELECT paid_users FROM paid_range)::float / (SELECT active_users FROM active_range)
    ELSE 0 END AS payment_conversion_rate,
  CASE WHEN (SELECT paid_users FROM paid_range) > 0
    THEN (SELECT repurchase_users FROM repurchase_range)::float / (SELECT paid_users FROM paid_range)
    ELSE 0 END AS repurchase_rate
`, qr.Start, qr.End, todayStart, todayEnd, monthStart, monthEnd).Scan(
		&out.TotalUsers,
		&out.DAU,
		&out.MAU,
		&out.TodayNewUsers,
		&out.TodayPaidUsers,
		&out.MonthRevenue,
		&out.ARPU,
		&out.PaymentConversionRate,
		&out.RepurchaseRate,
	)
	if err != nil {
		return nil, fmt.Errorf("query growth overview: %w", err)
	}
	return &out, nil
}
```

```go
func NewGrowthHandler(growthService *service.GrowthService) *GrowthHandler {

func (h *GrowthHandler) parseRange(c *gin.Context) (service.GrowthQueryRange, bool) {
	r, err := service.ParseGrowthQueryRange(
		c.Query("start_date"),
		c.Query("end_date"),
		c.Query("granularity"),
		time.Now(),
	)
	if err != nil {
		if errors.Is(err, service.ErrGrowthInvalidQueryRange) {
			response.BadRequest(c, err.Error())
			return service.GrowthQueryRange{}, false
		}
		response.Error(c, http.StatusBadRequest, err.Error())
		return service.GrowthQueryRange{}, false
	}
	return r, true
}

func (h *GrowthHandler) GetOverview(c *gin.Context) {
	r, ok := h.parseRange(c)
	if !ok {
		return
	}
	out, err := h.growthService.GetOverview(c.Request.Context(), r)
	if err != nil {
		response.InternalError(c, "Failed to get growth overview")
		return
	}
	response.Success(c, out)
}
```

```vue
<template>
  <AppLayout>
    <div class="space-y-6">
      <div class="flex flex-wrap items-start justify-between gap-4">
        <div>
          <h1 class="text-2xl font-semibold text-gray-900 dark:text-white">{{ t('admin.growth.title') }}</h1>
          <p class="mt-1 text-sm text-gray-500 dark:text-gray-400">{{ t('admin.growth.description') }}</p>
        </div>
        <div class="flex flex-wrap items-center gap-3">
          <DateRangePicker v-model:start-date="startDate" v-model:end-date="endDate" @change="onDateRangeChange" />
          <div class="w-28">
            <Select v-model="granularity" :options="granularityOptions" @change="reloadAll" />
          </div>
          <button class="btn btn-secondary" @click="reloadAll">{{ t('admin.growth.refresh') }}</button>
        </div>
      </div>

      <section class="space-y-4">
        <h2 class="text-base font-semibold text-gray-900 dark:text-white">{{ t('admin.growth.sections.overview') }}</h2>
        <GrowthKpiCards :overview="overview" />
      </section>

      <section class="grid grid-cols-1 gap-6 xl:grid-cols-2">
        <GrowthUserTrendChart :series="userTrend" :loading="loading.userTrend" />
        <GrowthRetentionMatrix :columns="retentionColumns" :cohorts="retentionCohorts" :loading="loading.retention" />
        <GrowthPaymentFunnel :steps="paymentFunnel" :tracking-ready="paymentTrackingReady" :loading="loading.payments" />
        <GrowthFeatureRanking :items="featureRanking" :session-metrics="sessionMetrics" :loading="loading.features" />
      </section>
    </div>
  </AppLayout>
</template>

<script setup lang="ts">
import { computed, onMounted, reactive, ref } from 'vue'
import { useI18n } from 'vue-i18n'
import { useAppStore } from '@/stores/app'
import { adminGrowthAPI, type GrowthGranularity, type GrowthOverview, type GrowthUserTrendPoint, type GrowthRetentionCohort, type GrowthFunnelStep, type GrowthFeatureRankingItem, type GrowthSessionMetrics } from '@/api/admin/growth'
import AppLayout from '@/components/layout/AppLayout.vue'
import DateRangePicker from '@/components/common/DateRangePicker.vue'
import Select from '@/components/common/Select.vue'
import GrowthKpiCards from '@/components/admin/growth/GrowthKpiCards.vue'
import GrowthUserTrendChart from '@/components/admin/growth/GrowthUserTrendChart.vue'
import GrowthRetentionMatrix from '@/components/admin/growth/GrowthRetentionMatrix.vue'
import GrowthPaymentFunnel from '@/components/admin/growth/GrowthPaymentFunnel.vue'
import GrowthFeatureRanking from '@/components/admin/growth/GrowthFeatureRanking.vue'
import { extractApiErrorMessage } from '@/utils/apiError'

const { t } = useI18n()
const appStore = useAppStore()

const formatDate = (date: Date) => `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, '0')}-${String(date.getDate()).padStart(2, '0')}`
const now = new Date()
const start = new Date(now)
start.setDate(start.getDate() - 29)

const startDate = ref(formatDate(start))
const endDate = ref(formatDate(now))
const granularity = ref<GrowthGranularity>('day')

const overview = ref<GrowthOverview | null>(null)
const userTrend = ref<GrowthUserTrendPoint[]>([])
const retentionColumns = ref<string[]>(['D1', 'D3', 'D7', 'D15', 'D30'])
const retentionCohorts = ref<GrowthRetentionCohort[]>([])
const paymentFunnel = ref<GrowthFunnelStep[]>([])
const paymentTrackingReady = ref(false)
const featureRanking = ref<GrowthFeatureRankingItem[]>([])
const sessionMetrics = ref<GrowthSessionMetrics | null>(null)

const loading = reactive({
  overview: false,
  userTrend: false,
  retention: false,
  payments: false,
  features: false
})

const granularityOptions = computed(() => [
  { value: 'day', label: t('admin.dashboard.day') },
  { value: 'week', label: t('admin.growth.week') },
  { value: 'month', label: t('admin.dashboard.month') }
])

const params = computed(() => ({
  start_date: startDate.value,
  end_date: endDate.value,
  granularity: granularity.value
}))

async function loadOverview() {
  loading.overview = true
  try {
    const res = await adminGrowthAPI.getOverview(params.value)
    overview.value = res.data
  } catch (err) {
    appStore.showError(extractApiErrorMessage(err, t('common.error')))
  } finally {
    loading.overview = false
  }
}

async function loadUserGrowth() {
  loading.userTrend = true
  try {
    const res = await adminGrowthAPI.getUserTrend(params.value)
    userTrend.value = res.data.series || []
  } catch (err) {
    appStore.showError(extractApiErrorMessage(err, t('common.error')))
  } finally {
    loading.userTrend = false
  }
}

async function loadRetention() {
  loading.retention = true
  try {
    const res = await adminGrowthAPI.getRetentionMatrix(params.value)
    retentionColumns.value = res.data.columns || ['D1', 'D3', 'D7', 'D15', 'D30']
    retentionCohorts.value = res.data.cohorts || []
  } catch (err) {
    appStore.showError(extractApiErrorMessage(err, t('common.error')))
  } finally {
    loading.retention = false
  }
}

async function loadPayments() {
  loading.payments = true
  try {
    const res = await adminGrowthAPI.getPaymentFunnel(params.value)
    paymentFunnel.value = res.data.steps || []
    paymentTrackingReady.value = res.data.tracking_ready === true
  } catch (err) {
    appStore.showError(extractApiErrorMessage(err, t('common.error')))
  } finally {
    loading.payments = false
  }
}

async function loadFeatures() {
  loading.features = true
  try {
    const [ranking, metrics] = await Promise.all([
      adminGrowthAPI.getFeatureRanking(params.value),
      adminGrowthAPI.getSessionMetrics(params.value)
    ])
    featureRanking.value = ranking.data.items || []
    sessionMetrics.value = metrics.data
  } catch (err) {
    appStore.showError(extractApiErrorMessage(err, t('common.error')))
  } finally {
    loading.features = false
  }
}

function reloadAll() {
  void Promise.all([loadOverview(), loadUserGrowth(), loadRetention(), loadPayments(), loadFeatures()])
}

function onDateRangeChange(range: { startDate: string; endDate: string }) {
  startDate.value = range.startDate
  endDate.value = range.endDate
  reloadAll()
}

onMounted(() => reloadAll())
</script>
```
